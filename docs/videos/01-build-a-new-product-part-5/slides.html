try {
  await client.connect(transport);
} catch (error) {
  await client.close().catch(() => undefined);
  // Never pass the server's message on: it can echo the request, credential included.
  throw new MagicPatternsUnavailable(/\b401\b|\b403\b|unauthori[sz]ed|invalid_token/i.test(String(error)) ? 'credential rejected' : 'unreachable');
}

worker/src/validators/mock-validator.ts
for (const width of WIDTHS) {
  await page.setViewportSize({ width, height: 800 });
  for (const path of renderable) {
    await page.goto(pathToFileURL(join(workspace, path)).href, { waitUntil: 'load' });
    const scrolls = await page.evaluate(
      () => document.documentElement.scrollWidth > document.documentElement.clientWidth,
    );
    if (scrolls) problems.push(`${path} scrolls horizontally at ${width} px`);
  }
}
